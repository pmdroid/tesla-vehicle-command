import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid>
	<Card title="Pair once" icon="approve-check">
		Load or create a P-256 key. Whitelist it with `Authenticator::BuildKeyWhitelistMessage`. Present an NFC card if the car returns `WAIT`.
	</Card>
	<Card title="Two domains" icon="information">
		Request session info on VCSEC and Infotainment. `Session::UpdateSessionInfo` checks the HMAC, then `BuildRoutableMessage` encrypts the payload.
	</Card>
	<Card title="Encode, wrap, write" icon="puzzle">
		`CarServer::*` and `Security::*` encode Tesla actions. `Session::BuildRoutableMessage` encrypts them. `BleFrame` reassembles indications. You write the GATT characteristic.
	</Card>
	<Card title="Pinned to Tesla" icon="github">
		Protos from teslamotors/vehicle-command `f97fa1e`. Host tests include Tesla protocol.md vectors and Go `proto.Marshal` goldens.
	</Card>
</CardGrid>
